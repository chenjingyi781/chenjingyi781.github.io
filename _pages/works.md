<script>
(function(){
  var lb = document.getElementById('wLightbox');
  if (!lb) return;
  var imgEl = lb.querySelector('.w-lb-img');
  var capEl = lb.querySelector('.w-lb-cap');
  var cards = Array.prototype.slice.call(document.querySelectorAll('.wpage .w-card'));
  var items = cards.map(function(c){
    var im = c.querySelector('img');
    return { src: im ? im.getAttribute('src') : '', cap: c.getAttribute('data-cap') || '' };
  });
  var cur = 0;
  function show(i){
    cur = (i + items.length) % items.length;
    imgEl.src = items[cur].src;
    capEl.textContent = items[cur].cap;
  }
  function open(i){ show(i); lb.classList.add('open'); document.body.style.overflow = 'hidden'; }
  function close(){ lb.classList.remove('open'); document.body.style.overflow = ''; }
  cards.forEach(function(c, i){
    c.addEventListener('click', function(){ open(i); });
  });
  lb.querySelector('.w-lb-close').addEventListener('click', close);
  lb.querySelector('.w-lb-prev').addEventListener('click', function(){ show(cur - 1); });
  lb.querySelector('.w-lb-next').addEventListener('click', function(){ show(cur + 1); });
  lb.addEventListener('click', function(e){ if (e.target === lb) close(); });
  document.addEventListener('keydown', function(e){
    if (!lb.classList.contains('open')) return;
    if (e.key === 'Escape') close();
    if (e.key === 'ArrowLeft') show(cur - 1);
    if (e.key === 'ArrowRight') show(cur + 1);
  });
})();
</script>
